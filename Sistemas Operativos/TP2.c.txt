#include <stdio.h> //padrão
#include <stdlib.h> //padrão
#include <unistd.h> //para forks , pids, execlp, exit e fprintf
#include <sys/types.h> //para o pid_t
#include <sys/wait.h> //para o waitpid

int main(int argc, char *argv[]) {

    int num_comandos;

    if (argc < 2) { // o valor mínimo é sempre 1 porque o próprio programa é um argumento
        fprintf(stdout, "Uso: %s <comando1> <comando2> ... <comandoX>\n", argv[0]);
        exit(1);
    }

    num_comandos = argc - 1; // o nome do programa conta como argumento então fazemos -1, o número de comandos é uma variável que armazena o número de
                             // argumentos.

    printf("LISTA DE ARGUMENTOS:\n"); //lista de argumentos = número de comandos
    for (int i = 1; i <= num_comandos; i++) {
        printf("%d. %s\n", i, argv[i]);//printf ao número do argumento, o argumento e \n para parágrafo
    }

    //por cada argumento criar um filho com fork(), o número de filhos(fork()) = número de argumentos
    for (int i = 1; i <= num_comandos; i++) {
        pid_t pid = fork();

        if (pid == -1) {
            perror("Erro ao criar processo filho");
            exit(1);
        } else if (pid == 0) {
            // Código do processo filho
            printf("\nA processar filho com PID=%d & PID PAI=%d...\n", getpid(), getppid());
            if (execlp(argv[i], argv[i], NULL) == -1) { // execlp executa os argumentos
                fprintf(stdout, "\nO comando %s nao foi concluido com sucesso\n", argv[i]);
                exit(1);
            }
        }
  }

    // O Processo pai vai aguardar a conclusão de cada processo filho
    int num_completos = 0; // número de processos filhos que foram completados com sucesso
    while (num_completos < num_comandos) {
        int status;
        pid_t child_pid = waitpid(-1, &status, 0); //waitpid espera pelos processos,o status armazena a saída do processo filho, o 0 bloqueia o processo pai até que os filhos terminem.

        if (child_pid == -1) { //-1 indica que o processo falhou ou ocorreu um erro
            perror("Erro ao esperar pelo processo filho");
            exit(1);
        }

        if (status == 0) {
        printf("\nProcesso filho %d terminou com sucesso\n", child_pid); // O valor zero em status indica que o processo filho terminou normalmente.
    	} 
	else {
        fprintf(stderr, "\nProcesso filho %d terminou mal ou com código de saída diferente de zero\n", child_pid); // Um valor diferente de zero em status indica que o processo filho terminou com um código de saída diferente de zero.
	}

        num_completos++;
    }

    printf("\nTUDO TERMINADO, FIM!\n");

    return 0;
}
